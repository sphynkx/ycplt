# Auto-detects the encoding of .txt/.html source files in rag_data/
# (cp1251, koi8-r, UTF-16, etc.) so build_index.py doesn't require every
# file to be UTF-8 — without it, build_index.py falls back to a fixed
# encoding list, which is less reliable but still works.
charset-normalizer
# Strips tags/scripts/styles to read .html/.htm source files.
beautifulsoup4
# Converts .rtf source files to plain text — pure Python, no system tool needed.
striprtf
# Extracts .rar/.arj/.7z source archives (build_index.py reads .zip
# directly via the stdlib, no extra package needed for that one). Also
# needs a matching system tool installed — see README.md's "RAG source
# formats" section for exact package names on Debian/Ubuntu and Fedora
# (they differ, and Fedora's own packaging for this has been in flux).
patool
# Legacy binary .doc (MS Word 97-2003) source files are read via the
# antiword system tool if present — no Python package needed for this
# one; without antiword installed, build_index.py falls back to a cruder
# built-in text scrape instead of skipping such files entirely.
#
# .ha archives (the old "HA" DOS-era format) have no current package on
# any distro and aren't recognized by patool at all — see build_index.py's
# module docstring. Not something pip/dnf/apt can fix; mentioned here so
# it isn't mistaken for an oversight in this file.

# Optional, only needed for the astrology chart tool (utils/astro.py):
# fully local/offline (Swiss Ephemeris via pyswisseph), no API key needed.
# AGPL-3.0 licensed — see utils/astro.py's module docstring. Also brings in
# pyswisseph itself as a dependency, which utils/electional.py imports
# directly too (as `swisseph`) for planetary-hour sunrise/sunset
# calculations (swe.rise_trans) — no separate pip entry needed for that,
# it's the same package kerykeion already requires.
kerykeion
# Resolves an IANA timezone name from lat/lon, fully offline (bundled
# timezone-boundary data) — lets the astro tool work without the user or
# model needing to know/state a timezone string explicitly.
timezonefinder
# Offline city-name -> coordinates/timezone lookup (~34k cities, bundled
# with the package, no separate download) — fallback for the astro tool
# when only a city name is given, no explicit coordinates.
geonamescache

# Optional, only needed for the "export message to PDF" button
# (utils/pdf_export.py, GET /api/messages/{id}/pdf): renders the message's
# HTML (text + embedded chart images) to a PDF via a real layout engine.
# Needs system libraries at runtime — Pango, cairo, gdk-pixbuf, and a font
# with Cyrillic + basic symbol coverage (DejaVu Sans, near-universal on
# Linux already). On Debian/Ubuntu:
#   apt install libpango-1.0-0 libpangocairo-1.0-0 libgdk-pixbuf-2.0-0 \
#     libcairo2 fonts-dejavu-core
# See weasyprint's own docs if `pip install weasyprint` succeeds but PDF
# generation fails at runtime — that's almost always one of these system
# libraries missing, not a Python-level problem.
weasyprint
